async function handleCreateSite(e) { e.preventDefault(); const name = document.getElementById('site-name').value.trim(); const subdomain = document.getElementById('site-subdomain').value.trim().toLowerCase(); const msg = document.getElementById('modal-msg'); const btn = document.getElementById('submit-site-btn'); btn.disabled = true; btn.innerText = 'Deploying...'; msg.className = 'text-xs text-center text-slate-400 min-h-[16px]'; msg.innerText = 'Creating site...'; try { const client = getSupabase(); const { error } = await client.from('projects').insert([ { user_id: currentUserId, name: name, site_name: name, subdomain: subdomain } ]); if (error) throw error; msg.className = 'text-xs text-center text-emerald-400 min-h-[16px]'; msg.innerText = '🎉 Site deployed successfully!'; setTimeout(() => { closeModal(); btn.disabled = false; btn.innerText = 'Deploy Site'; loadSites(); }, 1000); } catch (err) { msg.className = 'text-xs text-center text-red-400 min-h-[16px]'; msg.innerText = '❌ ' + (err.message || 'Error creating site'); btn.disabled = false; btn.innerText = 'Deploy Site'; } }
